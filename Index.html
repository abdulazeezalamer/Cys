<script>
const chatBox = document.getElementById('chat-box');
const userInput = document.getElementById('user-input');

function handleKey(e) {
if (e.key === 'Enter') sendMessage();
}

function sendMessage() {
const text = userInput.value.trim();
if (!text) return;

// إضافة رسالة المستخدم
appendMessage(text, 'user');
userInput.value = '';

// محاكاة رد النظام المؤقت (قبل ربط النموذج الفعلي)
setTimeout(() => {
const reply = `تم استلام الأمر: "${text}". (وضع المحاكاة الأولي نشط - جاهز للربط بالنموذج في المرحلة القادمة).`;
appendMessage(reply, 'ai');
}, 600);
}

function appendMessage(text, sender) {
const div = document.createElement('div');
div.className = `message ${sender === 'user' ? 'msg-user' : 'msg-ai'}`;

if (sender === 'ai') {
div.innerHTML = `
<div class="msg-header">SYSTEM // CYS-CORE</div>
${text}
<div class="feedback-actions">
<button class="btn-fb ok" onclick="rate(this, 'good')">✓ صحيح</button>
<button class="btn-fb bad" onclick="rate(this, 'bad')">✕ خطأ</button>
</div>
`;
} else {
div.textContent = text;
}

chatBox.appendChild(div);
chatBox.scrollTop = chatBox.scrollHeight;
}

function rate(btn, type) {
const parent = btn.parentElement;
parent.querySelectorAll('.btn-fb').forEach(b => b.classList.remove('active'));
btn.classList.add('active');
console.log(`Feedback logged: ${type}`);
}
</script>
